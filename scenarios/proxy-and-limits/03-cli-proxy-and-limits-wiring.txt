# Slice 03: CLI Proxy & Limits Wiring
# Covers PL-1, PL-2, PL-4, PL-8 (Operator limits on fetch-transcript, NO_PROXY host/port/IPv6 matching, per-hop redirect proxy switches, and --proxy overriding NO_PROXY on fetch-transcript)

# Operator DEFAULT_LIMIT and MAX_RESPONSE_BYTES apply to fetch-transcript too (PL-1, PL-2, D9, D25, I5)
> dns set www.youtube.com 142.250.190.78
dns: www.youtube.com -> 142.250.190.78

> env set DEFAULT_LIMIT 30
env: DEFAULT_LIMIT=30

> ytdlp available true
ytdlp: available true

> ytdlp serve https://www.youtube.com/watch?v=lim1 en <transcript><text start="0" dur="1">Longer transcript text here</text></transcript>
ytdlp: serve en https://www.youtube.com/watch?v=lim1

> fetch-transcript https://www.youtube.com/watch?v=lim1
| [Transcript language: en — en]

> env set DEFAULT_LIMIT notanumber
env: DEFAULT_LIMIT=notanumber

> fetch-transcript https://www.youtube.com/watch?v=lim1
error: Invalid DEFAULT_LIMIT "notanumber": must be a non-negative integer

> env clear
env: cleared

# MAX_RESPONSE_BYTES caps direct transcript fallback responses (PL-1, I5)
> env set MAX_RESPONSE_BYTES 50
env: MAX_RESPONSE_BYTES=50

> ytdlp available false
ytdlp: available false

> origin serve https://www.youtube.com/watch?v=bigyt 200 <html><script>var ytInitialPlayerResponse = {"captions":{"playerCaptionsTracklistRenderer":{"captionTracks":[]}}};</script></html>
origin: 200 https://www.youtube.com/watch?v=bigyt

> fetch-transcript https://www.youtube.com/watch?v=bigyt
error: Response too large: exceeded 50 byte limit

> env clear
env: cleared

# NO_PROXY port matching and explicit --proxy override on fetch-transcript (PL-4, PL-8, D6, D24, D26)
> env set HTTP_PROXY http://proxy.internal:8080
env: HTTP_PROXY=http://proxy.internal:8080

> env set NO_PROXY example.com:8080,www.youtube.com
env: NO_PROXY=example.com:8080,www.youtube.com

> dns set example.com 93.184.216.34
dns: example.com -> 93.184.216.34

> origin serve http://example.com:8080/p 200 <p>port-direct</p>
origin: 200 http://example.com:8080/p

> origin serve http://example.com:9090/p 200 <p>port-proxied</p>
origin: 200 http://example.com:9090/p

> fetch html http://example.com:8080/p
| <p>port-direct</p>

> origin last-proxy http://example.com:8080/p
proxy: direct

> fetch html http://example.com:9090/p
| <p>port-proxied</p>

> origin last-proxy http://example.com:9090/p
proxy: http://proxy.internal:8080/

> ytdlp available true
ytdlp: available true

> dns fail www.youtube.com ENOTFOUND
dns: fail www.youtube.com

> fetch-transcript https://www.youtube.com/watch?v=lim1 --proxy socks5://127.0.0.1:1080
| [Transcript language: en — en]
|
| [0:00] Longer transcript text here

> ytdlp last-args
args: --write-sub --write-auto-sub --sub-lang en --sub-format srv1 --skip-download -o <tmp>/sub --proxy socks5://127.0.0.1:1080 -- https://www.youtube.com/watch?v=lim1

# IPv6 NO_PROXY matching and per-hop redirect proxy transitions (direct -> proxied, http -> https proxy switch, proxied -> direct SSRF blocked) (PL-4, D24, I1)
> env clear
env: cleared

> env set HTTP_PROXY http://http-proxy.internal:8080
env: HTTP_PROXY=http://http-proxy.internal:8080

> env set HTTPS_PROXY https://https-proxy.internal:8443
env: HTTPS_PROXY=https://https-proxy.internal:8443

> env set NO_PROXY [2606:2800:220:1:248:1893:25c8:1946]:8080,direct.example.com
env: NO_PROXY=[2606:2800:220:1:248:1893:25c8:1946]:8080,direct.example.com

> dns set direct.example.com 93.184.216.34
dns: direct.example.com -> 93.184.216.34

> origin serve http://[2606:2800:220:1:248:1893:25c8:1946]:8080/v6 200 <p>v6-direct</p>
origin: 200 http://[2606:2800:220:1:248:1893:25c8:1946]:8080/v6

> origin serve http://[2606:2800:220:1:248:1893:25c8:1946]:9090/v6 200 <p>v6-proxied</p>
origin: 200 http://[2606:2800:220:1:248:1893:25c8:1946]:9090/v6

> fetch html http://[2606:2800:220:1:248:1893:25c8:1946]:8080/v6
| <p>v6-direct</p>

> origin last-proxy http://[2606:2800:220:1:248:1893:25c8:1946]:8080/v6
proxy: direct

> fetch html http://[2606:2800:220:1:248:1893:25c8:1946]:9090/v6
| <p>v6-proxied</p>

> origin last-proxy http://[2606:2800:220:1:248:1893:25c8:1946]:9090/v6
proxy: http://http-proxy.internal:8080/

> origin redirect http://direct.example.com/to-proxied 302 https://proxied.example.com/final
origin: redirect 302 http://direct.example.com/to-proxied -> https://proxied.example.com/final

> origin serve https://proxied.example.com/final 200 <p>switched-to-https-proxy</p>
origin: 200 https://proxied.example.com/final

> fetch html http://direct.example.com/to-proxied
| <p>switched-to-https-proxy</p>

> origin last-proxy http://direct.example.com/to-proxied
proxy: direct

> origin last-proxy https://proxied.example.com/final
proxy: https://https-proxy.internal:8443/

> origin redirect http://proxied.example.com/to-https 301 https://proxied.example.com/final
origin: redirect 301 http://proxied.example.com/to-https -> https://proxied.example.com/final

> fetch html http://proxied.example.com/to-https
| <p>switched-to-https-proxy</p>

> origin last-proxy http://proxied.example.com/to-https
proxy: http://http-proxy.internal:8080/

> origin last-proxy https://proxied.example.com/final
proxy: https://https-proxy.internal:8443/

> origin redirect https://proxied.example.com/to-ssrf 302 http://127.0.0.1/admin
origin: redirect 302 https://proxied.example.com/to-ssrf -> http://127.0.0.1/admin

> fetch html https://proxied.example.com/to-ssrf
error: Blocked Target URL: private address "127.0.0.1"
